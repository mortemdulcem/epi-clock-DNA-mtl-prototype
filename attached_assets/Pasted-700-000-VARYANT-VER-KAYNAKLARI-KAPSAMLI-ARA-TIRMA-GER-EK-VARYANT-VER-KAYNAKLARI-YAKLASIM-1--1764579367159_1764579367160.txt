def download_1000genomes_variants():
    """
    1000 Genomes varyantlarını indir ve filtrele
    """
    base_url = "ftp://ftp.1000genomes.ebi.ac.uk/vol1/ftp/release/20130502/"
    
    # Kromozom 1-22 + X
    for chrom in list(range(1, 23)) + ['X']:
        vcf_file = f"ALL.chr{chrom}.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes.vcf.gz"
        url = base_url + vcf_file
        
        print(f"Downloading chromosome {chrom}...")
        subprocess.run(['wget', '-c', url])  # -c: continue if interrupted
        
    print("Download complete!")

# Varyantları filtrele (sadece yaygın varyantlar, MAF > 0.01)
def filter_common_variants(input_vcf, output_vcf, maf_threshold=0.01):
    """
    Yaygın varyantları filtrele
    """
    cmd = f"""
    bcftools view -q {maf_threshold}:minor \
        {input_vcf} \
        -Oz -o {output_vcf}
    """
    subprocess.run(cmd, shell=True)
    
    # Index oluştur
    subprocess.run(f"bcftools index {output_vcf}", shell=True)

# Python ile gnomAD indirme ve filtreleme
import subprocess
import pandas as pd
from google.cloud import storage

def download_gnomad_variants(chromosomes=['1', '2', '3'], output_dir='gnomad_data'):
    """
    gnomAD varyantlarını Google Cloud'dan indir
    
    Args:
        chromosomes: İndirilecek kromozomlar (liste)
        output_dir: Çıktı dizini
    """
    bucket_name = 'gcp-public-data--gnomad'
    prefix = 'release/4.0/vcf/exomes/'
    
    # Google Cloud Storage client
    client = storage.Client.create_anonymous_client()
    bucket = client.bucket(bucket_name)
    
    for chrom in chromosomes:
        vcf_file = f'gnomad.exomes.v4.0.sites.chr{chrom}.vcf.bgz'
        tbi_file = f'{vcf_file}.tbi'
        
        print(f"Downloading chromosome {chrom}...")
        
        # VCF indir
        blob_vcf = bucket.blob(f'{prefix}{vcf_file}')
        blob_vcf.download_to_filename(f'{output_dir}/{vcf_file}')
        
        # Index indir
        blob_tbi = bucket.blob(f'{prefix}{tbi_file}')
        blob_tbi.download_to_filename(f'{output_dir}/{tbi_file}')
        
        print(f"  Chromosome {chrom} downloaded!")

# Bağımlılık-ilişkili genleri filtrele
def filter_addiction_genes(input_vcf, output_vcf):
    """
    Bağımlılık-ilişkili genlerdeki varyantları filtrele
    """
    # Bağımlılık genleri listesi
    addiction_genes = [
        'OPRM1', 'OPRD1', 'OPRK1',  # Opioid reseptörleri
        'DRD1', 'DRD2', 'DRD3', 'DRD4', 'DRD5',  # Dopamin reseptörleri
        'SLC6A3', 'SLC6A4',  # DAT, SERT
        'HTR1A', 'HTR1B', 'HTR2A',  # Serotonin reseptörleri
        'COMT', 'MAOA', 'MAOB',  # Metabolizma
        'GABRA1', 'GABRA2', 'GABRG2',  # GABA reseptörleri
        'CHRNA5', 'CHRNA3', 'CHRNB4',  # Nikotin reseptörleri
        'ADH1B', 'ADH1C', 'ALDH2',  # Alkol metabolizması
        'CYP2D6', 'CYP3A4', 'CYP2C19',  # İlaç metabolizması
        'BDNF', 'CREB1', 'FOS',  # Nöroplastisite
    ]
    
    # Gene listesini dosyaya yaz
    with open('addiction_genes.txt', 'w') as f:
        for gene in addiction_genes:
            f.write(f"{gene}\n")
    
    # VEP ile genlere göre filtrele
    cmd = f"""
    bcftools view {input_vcf} | \
    vep --cache --format vcf --force_overwrite \
        --gene_list addiction_genes.txt \
        --output_file {output_vcf} \
        --vcf
    """
    subprocess.run(cmd, shell=True)

# UK Biobank veri erişimi (başvuru sonrası)
def access_ukbiobank_data(application_id, key_file):
    """
    UK Biobank verilerine erişim
    
    Args:
        application_id: Onaylanan başvuru ID'si
        key_file: Güvenlik anahtarı dosyası
    """
    import ukbiobank
    
    # Bağlan
    ukb = ukbiobank.UKBiobank(application_id, key_file)
    
    # Genetik veri indir
    ukb.download_genetics(
        chromosomes=range(1, 23),
        output_format='bgen',
        output_dir='ukb_genetics'
    )
    
    # BGEN'i VCF'e çevir
    for chrom in range(1, 23):
        subprocess.run(f"""
            plink2 --bgen ukb_genetics/chr{chrom}.bgen \
                   --sample ukb_genetics/chr{chrom}.sample \
                   --export vcf bgz \
                   --out ukb_chr{chrom}
        """, shell=True)

    Genetik veri ile fenotip verilerini birleştir
    """
    # Fenotip yükle
    pheno = pd.read_csv(phenotype_file, sep='\t')
    
    # Madde kullanımı değişkenleri
    substance_fields = [
        '20117',  # Alcohol intake frequency
        '20160',  # Ever smoked
        '20453',  # Cannabis use
        '20454',  # Opioid use
    ]
    
    # Filtrele
    substance_users = pheno[pheno[substance_fields].notna().any(axis=1)]
    
    return substance_users

# ADIM 2: GWAS CATALOG'DAN BAĞIMLILIK-İLİŞKİLİ VARYANTLARI İNDİR
# ============================================================================

def download_gwas_addiction_variants():
    """
    GWAS Catalog'dan bağımlılık-ilişkili varyantları indir
    """
    # GWAS Catalog API
    base_url = "https://www.ebi.ac.uk/gwas/rest/api/studies"
    
    # Bağımlılık-ilişkili trait'ler
    addiction_traits = [
        'alcohol dependence',
        'opioid dependence',
        'cocaine dependence',
        'nicotine dependence',
        'cannabis use',
        'substance use disorder',
        'smoking behavior',
        'alcohol consumption',
        'drug addiction',
    ]
    
    all_variants = []
    
    for trait in addiction_traits:
        print(f"Downloading GWAS variants for: {trait}")
        
        # API sorgusu
        response = requests.get(
            f"{base_url}/search/findByDiseaseTrait",
            params={'q': trait}
        )
        
        if response.status_code == 200:
            studies = response.json()['_embedded']['studies']
            
            for study in studies:
                study_id = study['accessionId']
                
                # Varyantları indir
                var_response = requests.get(
                    f"https://www.ebi.ac.uk/gwas/rest/api/studies/{study_id}/associations"
                )
                
                if var_response.status_code == 200:
                    associations = var_response.json()['_embedded']['associations']
                    
                    for assoc in associations:
                        if 'loci' in assoc:
                            for locus in assoc['loci']:
                                if 'strongestRiskAlleles' in locus:
                                    for allele in locus['strongestRiskAlleles']:
                                        variant_info = {
                                            'rsid': allele.get('riskAlleleName', '').split('-')[0],
                                            'trait': trait,
                                            'pvalue': assoc.get('pvalue'),
                                            'or': assoc.get('orPerCopyNum'),
                                            'study': study_id
                                        }
                                        all_variants.append(variant_info)
    
    df_gwas = pd.DataFrame(all_variants)
    df_gwas = df_gwas.drop_duplicates(subset=['rsid'])
    
    print(f"GWAS Catalog'dan indirilen varyant sayısı: {len(df_gwas)}")
    
    # Kaydet
    df_gwas.to_csv('gwas_addiction_variants.csv', index=False)
    
    return df_gwas

# ADIM 3: 1000 GENOMES'TAN GEN BÖLGELERİNDEKİ VARYANTLARI FİLTRELE
# ============================================================================

def extract_variants_from_genes(gene_list, output_file='addiction_panel_700k.vcf.gz'):
    """
    1000 Genomes'tan gen bölgelerindeki tüm varyantları çıkar
    """
    # Gen koordinatlarını al (GENCODE)
    print("Downloading gene coordinates from GENCODE...")
    gencode_url = "https://ftp.ebi.ac.uk/pub/databases/gencode/Gencode_human/release_44/gencode.v44.annotation.gtf.gz"
    subprocess.run(f"wget -O gencode.gtf.gz {gencode_url}", shell=True)
    
    # GTF'i parse et
    import gzip
    gene_coords = {}
    
    with gzip.open('gencode.gtf.gz', 'rt') as f:
        for line in f:
            if line.startswith('#'):
                continue
            
            fields = line.strip().split('\t')
            if fields[2] == 'gene':
                # Gene name çıkar
                attrs = dict(item.strip().split(' ', 1) for item in fields[8].split(';') if item.strip())
                gene_name = attrs.get('gene_name', '').strip('"')
                
                if gene_name in gene_list:
                    chrom = fields[0].replace('chr', '')
                    start = int(fields[3])
                    end = int(fields[4])
                    
                    # ±50kb padding ekle (regulatory regions)
                    gene_coords[gene_name] = {
                        'chrom': chrom,
                        'start': max(1, start - 50000),
                        'end': end + 50000
                    }
    
    print(f"Found coordinates for {len(gene_coords)} genes")
    
    # BED dosyası oluştur
    with open('addiction_genes.bed', 'w') as f:
        for gene, coords in gene_coords.items():
            f.write(f"{coords['chrom']}\t{coords['start']}\t{coords['end']}\t{gene}\n")
    
    # 1000 Genomes'tan varyantları çıkar
    print("Extracting variants from 1000 Genomes...")
    
    base_url = "ftp://ftp.1000genomes.ebi.ac.uk/vol1/ftp/release/20130502/"
    
    for chrom in list(range(1, 23)) + ['X']:
        vcf_file = f"ALL.chr{chrom}.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes.vcf.gz"
        
        print(f"Processing chromosome {chrom}...")
        
        # İndir (yoksa)
        if not Path(vcf_file).exists():
            subprocess.run(f"wget {base_url}{vcf_file}", shell=True)
            subprocess.run(f"wget {base_url}{vcf_file}.tbi", shell=True)
        
        # BED bölgelerini filtrele
        bed_chrom = f"addiction_genes_chr{chrom}.bed"
        subprocess.run(f"grep '^{chrom}\t' addiction_genes.bed > {bed_chrom}", shell=True)
        
        # bcftools ile çıkar
        output_chrom = f"addiction_variants_chr{chrom}.vcf.gz"
        subprocess.run(f"""
            bcftools view -R {bed_chrom} {vcf_file} -Oz -o {output_chrom}
            bcftools index {output_chrom}
        """, shell=True)
    
    # Tüm kromozomları birleştir
    print("Merging all chromosomes...")
    chr_files = [f"addiction_variants_chr{c}.vcf.gz" for c in list(range(1, 23)) + ['X']]
    subprocess.run(f"""
        bcftools concat {' '.join(chr_files)} -Oz -o {output_file}
        bcftools index {output_file}
    """, shell=True)
    
    # Varyant sayısını say
    result = subprocess.run(
        f"bcftools view -H {output_file} | wc -l",
        shell=True,
        capture_output=True,
        text=True
    )
    n_variants = int(result.stdout.strip())
    
    print(f"Total variants extracted: {n_variants:,}")
    
    return output_file, n_variants

# ADIM 4: GWAS + GEN-BAZLI VARYANTLARI BİRLEŞTİR
# ============================================================================

def merge_gwas_and_gene_variants(gwas_file, gene_vcf, output_file='final_addiction_panel_700k.vcf.gz'):
    """
    GWAS ve gen-bazlı varyantları birleştir
    """
    # GWAS rsID'lerini çıkar
    gwas_df = pd.read_csv(gwas_file)
    rsids = gwas_df['rsid'].dropna().unique()
    
    # rsID listesini dosyaya yaz
    with open('gwas_rsids.txt', 'w') as f:
        for rsid in rsids:
            f.write(f"{rsid}\n")
    
    # 1000 Genomes'tan GWAS varyantlarını çıkar
    print("Extracting GWAS variants from 1000 Genomes...")
    
    base_url = "ftp://ftp.1000genomes.ebi.ac.uk/vol1/ftp/release/20130502/"
    gwas_vcfs = []
    
    for chrom in list(range(1, 23)) + ['X']:
        vcf_file = f"ALL.chr{chrom}.phase3_shapeit2_mvncall_integrated_v5b.20130502.genotypes.vcf.gz"
        
        if not Path(vcf_file).exists():
            subprocess.run(f"wget {base_url}{vcf_file}", shell=True)
            subprocess.run(f"wget {base_url}{vcf_file}.tbi", shell=True)
        
        output_gwas = f"gwas_variants_chr{chrom}.vcf.gz"
        subprocess.run(f"""
            bcftools view -i 'ID=@gwas_rsids.txt' {vcf_file} -Oz -o {output_gwas}
            bcftools index {output_gwas}
        """, shell=True)
        
        gwas_vcfs.append(output_gwas)
    
    # GWAS varyantlarını birleştir
    subprocess.run(f"""
        bcftools concat {' '.join(gwas_vcfs)} -Oz -o gwas_all.vcf.gz
        bcftools index gwas_all.vcf.gz
    """, shell=True)
    
    # Gen-bazlı ve GWAS varyantlarını birleştir
    print("Merging gene-based and GWAS variants...")
    subprocess.run(f"""
        bcftools concat -a {gene_vcf} gwas_all.vcf.gz -Oz -o {output_file}
        bcftools index {output_file}
    """, shell=True)
    
    # Tekrarları kaldır
    subprocess.run(f"""
        bcftools norm -d all {output_file} -Oz -o {output_file}.dedup.vcf.gz
        mv {output_file}.dedup.vcf.gz {output_file}
        bcftools index -f {output_file}
    """, shell=True)
    
    # Final sayım
    result = subprocess.run(
        f"bcftools view -H {output_file} | wc -l",
        shell=True,
        capture_output=True,
        text=True
    )
    n_variants = int(result.stdout.strip())
    
    print(f"\n{'='*80}")
    print(f"FINAL ADDICTION PANEL CREATED")
    print(f"{'='*80}")
    print(f"Total variants: {n_variants:,}")
    print(f"Output file: {output_file}")
    print(f"File size: {Path(output_file).stat().st_size / 1024 / 1024:.1f} MB")
    
    return output_file, n_variants

# ADIM 5: VARYANT PANELİNİ OPTİMİZE ET (TAM 700K)
# ============================================================================

def optimize_to_700k(input_vcf, output_vcf='addiction_panel_700k_optimized.vcf.gz', target_n=700000):
    """
    Varyant panelini tam 700K'ya optimize et
    
    Öncelik sırası:
    1. GWAS significant variants (p<5e-8)
    2. Exonic variants (missense, nonsense, frameshift)
    3. Regulatory variants (promoter, enhancer)
    4. Common variants (MAF > 0.05)
    5. Rare functional variants (MAF < 0.01, CADD > 15)
    """
    print("Optimizing variant panel to 700K...")
    
    # Mevcut varyant sayısını kontrol et
    result = subprocess.run(
        f"bcftools view -H {input_vcf} | wc -l",
        shell=True,
        capture_output=True,
        text=True
    )
    current_n = int(result.stdout.strip())
    
    print(f"Current variants: {current_n:,}")
    print(f"Target: {target_n:,}")
    
    if current_n < target_n:
        print(f"Need to add {target_n - current_n:,} more variants")
        
        # Ek varyantlar ekle (yaygın varyantlar)
        subprocess.run(f"""
            bcftools view -q 0.05:minor {input_vcf} | \
            bcftools view -m2 -M2 -v snps | \
            head -n {target_n - current_n} | \
            bcftools view -Oz -o additional_variants.vcf.gz
            
            bcftools concat {input_vcf} additional_variants.vcf.gz -Oz -o {output_vcf}
            bcftools index {output_vcf}
        """, shell=True)
        
    elif current_n > target_n:
        print(f"Need to remove {current_n - target_n:,} variants")
        
        # Öncelikli varyantları seç
        subprocess.run(f"""
            # 1. GWAS variants (tümü)
            bcftools view -i 'ID!="."' {input_vcf} -Oz -o priority1.vcf.gz
            
            # 2. Exonic variants
            bcftools view -i 'INFO/ANN ~ "missense|nonsense|frameshift"' {input_vcf} -Oz -o priority2.vcf.gz
            
            # 3. Common variants
            bcftools view -q 0.05:minor {input_vcf} | head -n {target_n // 2} -Oz -o priority3.vcf.gz
            
            # Birleştir ve 700K'ya kırp
            bcftools concat -a priority1.vcf.gz priority2.vcf.gz priority3.vcf.gz | \
            bcftools norm -d all | \
            head -n {target_n} | \
            bcftools view -Oz -o {output_vcf}
            
            bcftools index {output_vcf}
        """, shell=True)
    
    else:
        print("Already at target size!")
        subprocess.run(f"cp {input_vcf} {output_vcf}", shell=True)
        subprocess.run(f"bcftools index {output_vcf}", shell=True)
    
    # Final doğrulama
    result = subprocess.run(
        f"bcftools view -H {output_vcf} | wc -l",
        shell=True,
        capture_output=True,
        text=True
    )
    final_n = int(result.stdout.strip())
    
    print(f"\n{'='*80}")
    print(f"OPTIMIZATION COMPLETE")
    print(f"{'='*80}")
    print(f"Final variant count: {final_n:,}")
    print(f"Target achieved: {'✓ YES' if abs(final_n - target_n) < 1000 else '✗ NO'}")
    
    return output_vcf, final_n

def create_addiction_700k_panel():
    """
    Tam workflow: 700K bağımlılık varyant paneli oluştur
    """
    print("=" * 80)
    print("CREATING ADDICTION-SPECIFIC 700K VARIANT PANEL")
    print("=" * 80)
    
    # Adım 1: Gen listesi
    print("\nStep 1: Defining addiction-related genes...")
    gene_list, gene_categories = get_addiction_genes()
    
    # Adım 2: GWAS varyantları
    print("\nStep 2: Downloading GWAS addiction variants...")
    gwas_df = download_gwas_addiction_variants()
    
    # Adım 3: Gen-bazlı varyantlar
    print("\nStep 3: Extracting variants from gene regions...")
    gene_vcf, n_gene_variants = extract_variants_from_genes(gene_list)
    
    # Adım 4: Birleştir
    print("\nStep 4: Merging GWAS and gene-based variants...")
    merged_vcf, n_merged = merge_gwas_and_gene_variants('gwas_addiction_variants.csv', gene_vcf)
    
    # Adım 5: 700K'ya optimize et
    print("\nStep 5: Optimizing to 700K variants...")
    final_vcf, n_final = optimize_to_700k(merged_vcf, target_n=700000)
    
    # Özet rapor
    print("\n" + "=" * 80)
    print("FINAL REPORT")
    print("=" * 80)
    print(f"Genes analyzed: {len(gene_list)}")
    print(f"GWAS variants: {len(gwas_df)}")
    print(f"Gene-region variants: {n_gene_variants:,}")
    print(f"Merged variants: {n_merged:,}")
    print(f"Final optimized: {n_final:,}")
    print(f"\nOutput file: {final_vcf}")
    print(f"File size: {Path(final_vcf).stat().st_size / 1024 / 1024:.1f} MB")
    
    # Varyant kategorileri
    print("\n" + "-" * 80)
    print("VARIANT CATEGORIES")
    print("-" * 80)
    
    for category, genes in gene_categories.items():
        print(f"{category:30s}: {len(genes):3d} genes")
    
    return final_vcf
