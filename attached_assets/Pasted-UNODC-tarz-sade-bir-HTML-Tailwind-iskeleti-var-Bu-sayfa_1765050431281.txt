  <!-- JS: SEKME GEÇİŞİ + BASİT AJAX İSKELETLERİ -->
  <script>
    // Sekme geçişleri
    const tabLinks = document.querySelectorAll('.tab-link');
    const tabContents = {
      upload: document.getElementById('tab-upload'),
      run: document.getElementById('tab-run'),
      export: document.getElementById('tab-export')
    };

    tabLinks.forEach(btn => {
      btn.addEventListener('click', () => {
        const target = btn.getAttribute('data-tab');
        tabLinks.forEach(b => {
          b.classList.remove('border-unblue', 'text-unblue', 'border-b-2');
          b.classList.add('text-slate-600');
        });
        btn.classList.remove('text-slate-600');
        btn.classList.add('border-b-2', 'border-unblue', 'text-unblue');
        Object.keys(tabContents).forEach(key => {
          tabContents[key].classList.remove('active');
        });
        tabContents[target].classList.add('active');
      });
    });

    // 1) DNA Dosya Yükleme (iskele: backend endpoint'ini sen belirleyeceksin)
    const uploadForm = document.getElementById('dna-upload-form');
    const uploadStatus = document.getElementById('upload-status');

    uploadForm.addEventListener('submit', async (e) => {
      e.preventDefault();
      const formData = new FormData(uploadForm);
      uploadStatus.textContent = 'Dosya yükleniyor...';

      try {
        const res = await fetch('/api/upload-dna', {
          method: 'POST',
          body: formData
        });
        if (!res.ok) throw new Error('Upload error');
        const data = await res.json();
        uploadStatus.textContent = `Yükleme tamamlandı. Veri seti ID: ${data.dataset_id || 'N/A'}`;
      } catch (err) {
        console.error(err);
        uploadStatus.textContent = 'Yükleme sırasında bir hata oluştu veya API henüz bağlı değil.';
        uploadStatus.classList.add('text-red-600');
      }
    });

    // 2) Analiz Başlatma
    const runForm = document.getElementById('run-analysis-form');
    const runStatus = document.getElementById('run-status');

    runForm.addEventListener('submit', async (e) => {
      e.preventDefault();
      runStatus.textContent = 'Analiz başlatılıyor...';
      const formData = new FormData(runForm);
      const payload = Object.fromEntries(formData.entries());

      try {
        const res = await fetch('/api/run-epiclock', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });
        if (!res.ok) throw new Error('Run error');
        const data = await res.json();
        runStatus.textContent = `Analiz başlatıldı. Çalışma ID: ${data.run_id || 'N/A'} · Durum: ${data.status || 'başlatıldı'}`;
      } catch (err) {
        console.error(err);
        runStatus.textContent = 'Analiz başlatılırken bir hata oluştu veya API henüz bağlı değil.';
        runStatus.classList.add('text-red-600');
      }
    });

    // 3) Çıktı / Rapor Alma
    const exportForm = document.getElementById('export-form');
    const exportStatus = document.getElementById('export-status');

    exportForm.addEventListener('submit', async (e) => {
      e.preventDefault();
      exportStatus.textContent = 'Çıktı hazırlanıyor...';
      const formData = new FormData(exportForm);
      const payload = Object.fromEntries(formData.entries());

      try {
        const res = await fetch('/api/export-epiclock', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });
        if (!res.ok) throw new Error('Export error');

        // Eğer çıktı dosya indirme linki ise:
        const data = await res.json();
        if (data.download_url) {
          exportStatus.innerHTML = `Çıktı hazır: <a href="${data.download_url}" class="text-unblue underline" target="_blank">İndir</a>`;
        } else {
          exportStatus.textContent = 'Çıktı alındı (API demo cevabı).';
        }
      } catch (err) {
        console.error(err);
        exportStatus.textContent = 'Çıktı alınırken bir hata oluştu veya API henüz bağlı değil.';
        exportStatus.classList.add('text-red-600');
      }
    });
  </script>
