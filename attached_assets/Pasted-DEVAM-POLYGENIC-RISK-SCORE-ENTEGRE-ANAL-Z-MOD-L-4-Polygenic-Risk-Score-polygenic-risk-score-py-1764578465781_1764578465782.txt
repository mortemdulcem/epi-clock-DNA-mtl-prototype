"""
Polygenic Risk Score (PRS) Module
Calculate genetic risk for substance use disorders
"""

import pandas as pd
import numpy as np
from typing import Dict, List, Tuple
from pathlib import Path

class PolygenicRiskCalculator:
    """
    Calculate Polygenic Risk Scores for addiction-related traits.
    
    Traits:
    - Alcohol dependence
    - Opioid dependence
    - Cocaine dependence
    - Nicotine dependence
    - Cannabis use disorder
    - General addiction liability
    
    Uses GWAS summary statistics from:
    - Psychiatric Genomics Consortium (PGC)
    - GWAS Catalog
    - UK Biobank
    """
    
    def __init__(self, gwas_dir="/data/gwas_sumstats"):
        self.gwas_dir = Path(gwas_dir)
        self.prs_weights = self._load_prs_weights()
        
    def calculate_prs(self, variants_df, trait='alcohol_dependence'):
        """
        Calculate PRS for a specific trait
        
        Args:
            variants_df: DataFrame with individual's variants
            trait: Trait name (alcohol_dependence, opioid_dependence, etc.)
            
        Returns:
            Dictionary with PRS and percentile
        """
        # Load trait-specific weights
        weights = self.prs_weights[trait]
        
        # Match variants
        matched = self._match_variants(variants_df, weights)
        
        # Calculate raw PRS
        raw_prs = self._calculate_raw_prs(matched)
        
        # Standardize (z-score)
        standardized_prs = self._standardize_prs(raw_prs, trait)
        
        # Convert to percentile
        percentile = self._prs_to_percentile(standardized_prs)
        
        # Interpret risk
        risk_category = self._interpret_risk(percentile)
        
        return {
            'trait': trait,
            'raw_prs': raw_prs,
            'standardized_prs': standardized_prs,
            'percentile': percentile,
            'risk_category': risk_category,
            'n_variants_used': len(matched),
            'interpretation': self._generate_interpretation(trait, percentile, risk_category)
        }
    
    def calculate_multi_trait_prs(self, variants_df):
        """
        Calculate PRS for all addiction-related traits
        
        Args:
            variants_df: DataFrame with individual's variants
            
        Returns:
            Dictionary with PRS for each trait
        """
        traits = [
            'alcohol_dependence',
            'opioid_dependence',
            'cocaine_dependence',
            'nicotine_dependence',
            'cannabis_use_disorder',
            'general_addiction_liability'
        ]
        
        results = {}
        for trait in traits:
            results[trait] = self.calculate_prs(variants_df, trait)
        
        # Calculate composite risk
        results['composite'] = self._calculate_composite_risk(results)
        
        return results
    
    def _load_prs_weights(self):
        """
        Load PRS weights from GWAS summary statistics
        
        Format:
        SNP | CHR | BP | A1 | A2 | BETA | P
        """
        weights = {}
        
        # Alcohol dependence (PGC + MVP)
        weights['alcohol_dependence'] = pd.read_csv(
            self.gwas_dir / 'alcohol_dependence_prs.txt',
            sep='\t'
        )
        
        # Opioid dependence (MVP)
        weights['opioid_dependence'] = pd.read_csv(
            self.gwas_dir / 'opioid_dependence_prs.txt',
            sep='\t'
        )
        
        # Cocaine dependence (PGC)
        weights['cocaine_dependence'] = pd.read_csv(
            self.gwas_dir / 'cocaine_dependence_prs.txt',
            sep='\t'
        )
        
        # Nicotine dependence (GSCAN)
        weights['nicotine_dependence'] = pd.read_csv(
            self.gwas_dir / 'nicotine_dependence_prs.txt',
            sep='\t'
        )
        
        # Cannabis use disorder
        weights['cannabis_use_disorder'] = pd.read_csv(
            self.gwas_dir / 'cannabis_use_disorder_prs.txt',
            sep='\t'
        )
        
        # General addiction liability
        weights['general_addiction_liability'] = pd.read_csv(
            self.gwas_dir / 'general_addiction_prs.txt',
            sep='\t'
        )
        
        return weights
    
    def _match_variants(self, variants_df, weights_df):
        """
        Match individual's variants with PRS weights
        
        Returns:
            DataFrame with matched variants and their weights
        """
        # Merge on SNP ID or position
        matched = variants_df.merge(
            weights_df,
            left_on='Uploaded_variation',
            right_on='SNP',
            how='inner'
        )
        
        # Extract dosage (0, 1, 2 copies of effect allele)
        matched['dosage'] = matched.apply(self._calculate_dosage, axis=1)
        
        return matched
    
    def _calculate_dosage(self, row):
        """
        Calculate allele dosage (0, 1, or 2)
        
        Genotype encoding:
        - 0/0 (ref/ref): 0 copies
        - 0/1 (ref/alt): 1 copy
        - 1/1 (alt/alt): 2 copies
        """
        # Parse genotype from VCF
        genotype = row.get('genotype', '0/0')
        
        if genotype == '0/0':
            return 0
        elif genotype in ['0/1', '1/0']:
            return 1
        elif genotype == '1/1':
            return 2
        else:
            return np.nan
    
    def _calculate_raw_prs(self, matched_df):
        """
        Calculate raw PRS
        
        PRS = Σ(beta_i × dosage_i)
        """
        prs = (matched_df['BETA'] * matched_df['dosage']).sum()
        return prs
    
    def _standardize_prs(self, raw_prs, trait):
        """
        Standardize PRS to z-score
        
        Uses population mean and SD from reference panel
        """
        # Load reference statistics
        ref_stats = self._load_reference_stats(trait)
        
        z_score = (raw_prs - ref_stats['mean']) / ref_stats['sd']
        return z_score
    
    def _load_reference_stats(self, trait):
        """
        Load population reference statistics
        
        From large reference panels (e.g., UK Biobank)
        """
        # Placeholder values (replace with actual reference data)
        reference = {
            'alcohol_dependence': {'mean': 0.0, 'sd': 1.0},
            'opioid_dependence': {'mean': 0.0, 'sd': 1.0},
            'cocaine_dependence': {'mean': 0.0, 'sd': 1.0},
            'nicotine_dependence': {'mean': 0.0, 'sd': 1.0},
            'cannabis_use_disorder': {'mean': 0.0, 'sd': 1.0},
            'general_addiction_liability': {'mean': 0.0, 'sd': 1.0}
        }
        return reference[trait]
    
    def _prs_to_percentile(self, z_score):
        """
        Convert z-score to percentile
        
        Uses standard normal distribution
        """
        from scipy.stats import norm
        percentile = norm.cdf(z_score) * 100
        return percentile
    
    def _interpret_risk(self, percentile):
        """
        Categorize risk based on percentile
        
        Categories:
        - Very Low: <10th percentile
        - Low: 10-25th percentile
        - Average: 25-75th percentile
        - High: 75-90th percentile
        - Very High: >90th percentile
        """
        if percentile < 10:
            return "Very Low"
        elif percentile < 25:
            return "Low"
        elif percentile < 75:
            return "Average"
        elif percentile < 90:
            return "High"
        else:
            return "Very High"
    
    def _generate_interpretation(self, trait, percentile, risk_category):
        """
        Generate clinical interpretation
        """
        trait_names = {
            'alcohol_dependence': 'Alcohol Dependence',
            'opioid_dependence': 'Opioid Dependence',
            'cocaine_dependence': 'Cocaine Dependence',
            'nicotine_dependence': 'Nicotine Dependence',
            'cannabis_use_disorder': 'Cannabis Use Disorder',
            'general_addiction_liability': 'General Addiction Liability'
        }
        
        interpretation = f"""
        {trait_names[trait]} Polygenic Risk Score:
        
        Percentile: {percentile:.1f}th
        Risk Category: {risk_category}
        
        """
        
        if risk_category in ["High", "Very High"]:
            interpretation += f"""
        CLINICAL IMPLICATIONS:
        - Increased genetic susceptibility to {trait_names[trait].lower()}
        - Enhanced monitoring recommended
        - Consider preventive interventions
        - Family history assessment important
        - Psychoeducation about genetic risk
        
        IMPORTANT: Genetic risk is only one factor. Environmental and 
        psychosocial factors play crucial roles in addiction development.
        """
        elif risk_category in ["Low", "Very Low"]:
            interpretation += f"""
        CLINICAL IMPLICATIONS:
        - Lower genetic susceptibility to {trait_names[trait].lower()}
        - Standard prevention strategies appropriate
        - Environmental factors remain important
        
        IMPORTANT: Low genetic risk does not eliminate possibility of 
        developing addiction. Behavioral and environmental factors are critical.
        """
        else:
            interpretation += f"""
        CLINICAL IMPLICATIONS:
        - Average genetic susceptibility to {trait_names[trait].lower()}
        - Standard clinical approach recommended
        - Focus on modifiable risk factors
        """
        
        return interpretation.strip()
    
    def _calculate_composite_risk(self, results):
        """
        Calculate composite addiction risk across all traits
        
        Uses weighted average of individual PRS
        """
        # Extract standardized PRS values
        prs_values = [
            results['alcohol_dependence']['standardized_prs'],
            results['opioid_dependence']['standardized_prs'],
            results['cocaine_dependence']['standardized_prs'],
            results['nicotine_dependence']['standardized_prs'],
            results['cannabis_use_disorder']['standardized_prs']
        ]
        
        # Calculate mean
        composite_prs = np.mean(prs_values)
        
        # Convert to percentile
        percentile = self._prs_to_percentile(composite_prs)
        
        # Interpret
        risk_category = self._interpret_risk(percentile)
        
        return {
            'composite_prs': composite_prs,
            'percentile': percentile,
            'risk_category': risk_category,
            'interpretation': f"""
            Composite Addiction Risk Score:
            
            Overall genetic susceptibility: {risk_category}
            Percentile: {percentile:.1f}th
            
            This score integrates genetic risk across multiple substance types,
            providing an overall assessment of addiction liability.
            """
        }
    
    def generate_report(self, results, output_file='prs_report.txt'):
        """
        Generate comprehensive PRS report
        """
        with open(output_file, 'w') as f:
            f.write("=" * 80 + "\n")
            f.write("POLYGENIC RISK SCORE REPORT\n")
            f.write("Addiction-Related Traits\n")
            f.write("=" * 80 + "\n\n")
            
            # Individual traits
            for trait, data in results.items():
                if trait == 'composite':
                    continue
                f.write(f"\n{'-' * 80}\n")
                f.write(data['interpretation'])
                f.write(f"\n{'-' * 80}\n")
            
            # Composite
            f.write(f"\n{'=' * 80}\n")
            f.write("COMPOSITE RISK ASSESSMENT\n")
            f.write(f"{'=' * 80}\n")
            f.write(results['composite']['interpretation'])
            
            # Summary table
            f.write(f"\n\n{'=' * 80}\n")
            f.write("SUMMARY TABLE\n")
            f.write(f"{'=' * 80}\n\n")
            f.write(f"{'Trait':<40} {'Percentile':<15} {'Risk Category':<15}\n")
            f.write(f"{'-' * 80}\n")
            
            for trait, data in results.items():
                if trait == 'composite':
                    continue
                trait_name = trait.replace('_', ' ').title()
                f.write(f"{trait_name:<40} {data['percentile']:>6.1f}th        {data['risk_category']:<15}\n")
            
            f.write(f"{'-' * 80}\n")
            f.write(f"{'Composite Risk':<40} {results['composite']['percentile']:>6.1f}th        {results['composite']['risk_category']:<15}\n")
            
        print(f"Report saved to {output_file}")

"""
Integrated Analysis Module
Combine DNA methylation (EpiClock) + Genetic variants (PRS + PGx)
"""

import pandas as pd
import numpy as np
from typing import Dict, List
import matplotlib.pyplot as plt
import seaborn as sns

# Import EpiClock modules
from modules.epigenetic_clocks import EpigeneticClockCalculator
from modules.clinical_decision import ClinicalDecisionSupport

# Import new variant modules
from modules.variant_annotation import VariantAnnotator
from modules.pharmacogenomics import PharmacogenomicsAnalyzer
from modules.polygenic_risk_score import PolygenicRiskCalculator

class IntegratedAnalyzer:
    """
    Integrate multi-omics data for comprehensive addiction assessment.
    
    Combines:
    1. DNA Methylation (Epigenetic Age Acceleration)
    2. Genetic Variants (PRS, Pharmacogenomics)
    3. Clinical Data (Demographics, Substance Use History)
    
    Provides:
    - Comprehensive risk assessment
    - Personalized treatment recommendations
    - Prognostic predictions
    """
    
    def __init__(self):
        # Initialize sub-modules
        self.clock_calculator = EpigeneticClockCalculator()
        self.clinical_support = ClinicalDecisionSupport()
        self.variant_annotator = VariantAnnotator()
        self.pgx_analyzer = PharmacogenomicsAnalyzer()
        self.prs_calculator = PolygenicRiskCalculator()
        
    def analyze_patient(self, 
                       methylation_data,
                       vcf_file,
                       clinical_data):
        """
        Full integrated analysis for a single patient
        
        Args:
            methylation_data: DNA methylation beta values (DataFrame or file path)
            vcf_file: Variant call format file (path)
            clinical_data: Dictionary with clinical information
            
        Returns:
            Comprehensive analysis results
        """
        results = {}
        
        # === 1. EPIGENETIC AGE ANALYSIS ===
        print("Step 1/5: Calculating epigenetic ages...")
        epi_results = self._analyze_epigenetic_age(methylation_data, clinical_data)
        results['epigenetic'] = epi_results
        
        # === 2. VARIANT ANNOTATION ===
        print("Step 2/5: Annotating genetic variants...")
        variants_df = self.variant_annotator.annotate(vcf_file, "temp_annotated.tsv")
        results['variants'] = {
            'n_total': len(variants_df),
            'n_functional': len(variants_df[variants_df['IMPACT'] == 'HIGH']),
            'data': variants_df
        }
        
        # === 3. PHARMACOGENOMICS ===
        print("Step 3/5: Analyzing pharmacogenomic variants...")
        pgx_results = self.pgx_analyzer.analyze(variants_df)
        results['pharmacogenomics'] = pgx_results
        
        # === 4. POLYGENIC RISK SCORES ===
        print("Step 4/5: Calculating polygenic risk scores...")
        prs_results = self.prs_calculator.calculate_multi_trait_prs(variants_df)
        results['prs'] = prs_results
        
        # === 5. INTEGRATED RISK ASSESSMENT ===
        print("Step 5/5: Generating integrated risk assessment...")
        integrated_risk = self._calculate_integrated_risk(results, clinical_data)
        results['integrated_risk'] = integrated_risk
        
        # === 6. PERSONALIZED RECOMMENDATIONS ===
        recommendations = self._generate_recommendations(results, clinical_data)
        results['recommendations'] = recommendations
        
        return results
    
    def _analyze_epigenetic_age(self, methylation_data, clinical_data):
        """
        Analyze epigenetic age acceleration
        """
        # Calculate all clocks
        ages = self.clock_calculator.calculate_all_clocks(methylation_data)
        
        # Calculate acceleration
        chronological_age = clinical_data['age']
        acceleration = {}
        
        for clock, epi_age in ages.items():
            acceleration[clock] = epi_age - chronological_age
        
        # Clinical interpretation
        interpretation = self.clinical_support.interpret_acceleration(
            acceleration,
            clinical_data
        )
        
        return {
            'chronological_age': chronological_age,
            'epigenetic_ages': ages,
            'acceleration': acceleration,
            'mean_acceleration': np.mean(list(acceleration.values())),
            'interpretation': interpretation
        }
    
    def _calculate_integrated_risk(self, results, clinical_data):
        """
        Calculate integrated risk score combining all data sources
        
        Risk factors:
        1. Epigenetic age acceleration (weight: 0.3)
        2. Polygenic risk score (weight: 0.3)
        3. Pharmacogenomic risk variants (weight: 0.2)
        4. Clinical history (weight: 0.2)
        """
        # 1. Epigenetic component
        epi_acceleration = results['epigenetic']['mean_acceleration']
        epi_risk = self._normalize_risk(epi_acceleration, min_val=-5, max_val=10)
        
        # 2. Genetic component (PRS)
        composite_prs = results['prs']['composite']['percentile']
        prs_risk = composite_prs / 100  # Convert to 0-1 scale
        
        # 3. Pharmacogenomic component
        pgx_risk = self._assess_pgx_risk(results['pharmacogenomics'])
        
        # 4. Clinical component
        clinical_risk = self._assess_clinical_risk(clinical_data)
        
        # Weighted combination
        integrated_risk_score = (
            0.3 * epi_risk +
            0.3 * prs_risk +
            0.2 * pgx_risk +
            0.2 * clinical_risk
        )
        
        # Convert to percentile
        risk_percentile = integrated_risk_score * 100
        
        # Categorize
        if risk_percentile < 25:
            risk_category = "Low"
        elif risk_percentile < 50:
            risk_category = "Moderate"
        elif risk_percentile < 75:
            risk_category = "High"
        else:
            risk_category = "Very High"
        
        return {
            'integrated_risk_score': integrated_risk_score,
            'risk_percentile': risk_percentile,
            'risk_category': risk_category,
            'components': {
                'epigenetic': epi_risk,
                'genetic_prs': prs_risk,
                'pharmacogenomic': pgx_risk,
                'clinical': clinical_risk
            },
            'interpretation': self._interpret_integrated_risk(
                risk_percentile, 
                risk_category,
                results
            )
        }
    
    def _normalize_risk(self, value, min_val, max_val):
        """Normalize value to 0-1 scale"""
        normalized = (value - min_val) / (max_val - min_val)
        return np.clip(normalized, 0, 1)
    
    def _assess_pgx_risk(self, pgx_results):
        """
        Assess pharmacogenomic risk
        
        High-risk factors:
        - CYP2D6 poor metabolizer
        - DRD2 Taq1A A1/A1 (high addiction risk)
        - COMT Val/Val (poor treatment response)
        """
        risk_score = 0.5  # Baseline
        
        # Opioid metabolism
        if pgx_results['opioid']['cyp2d6_phenotype'] == "Poor Metabolizer (PM)":
            risk_score += 0.2
        elif pgx_results['opioid']['cyp2d6_phenotype'] == "Ultra-rapid Metabolizer (UM)":
            risk_score += 0.15
        
        # Addiction risk (DRD2)
        addiction_risk = pgx_results['stimulant']['addiction_risk']
        if addiction_risk == "INCREASED":
            risk_score += 0.2
        elif addiction_risk == "MODERATE":
            risk_score += 0.1
        
        # Alcohol protection (ALDH2)
        alcohol_risk = pgx_results['alcohol']['alcohol_dependence_risk']
        if alcohol_risk == "LOW":
            risk_score -= 0.2
        
        return np.clip(risk_score, 0, 1)
    
    def _assess_clinical_risk(self, clinical_data):
        """
        Assess clinical risk based on history
        
        Risk factors:
        - Substance use duration
        - Number of substances
        - Family history
        - Age of onset
        - Previous treatment failures
        """
        risk_score = 0.0
        
        # Duration of use
        if 'substance_use_years' in clinical_data:
            years = clinical_data['substance_use_years']
            if years > 10:
                risk_score += 0.3
            elif years > 5:
                risk_score += 0.2
            elif years > 2:
                risk_score += 0.1
        
        # Polysubstance use
        if 'n_substances' in clinical_data:
            n_sub = clinical_data['n_substances']
            if n_sub >= 3:
                risk_score += 0.3
            elif n_sub == 2:
                risk_score += 0.2
        
        # Family history
        if clinical_data.get('family_history_addiction', False):
            risk_score += 0.2
        
        # Early onset
        if 'age_of_onset' in clinical_data:
            if clinical_data['age_of_onset'] < 15:
                risk_score += 0.3
            elif clinical_data['age_of_onset'] < 18:
                risk_score += 0.2
        
        # Treatment failures
        if 'n_treatment_failures' in clinical_data:
            failures = clinical_data['n_treatment_failures']
            risk_score += min(0.3, failures * 0.1)
        
        return np.clip(risk_score, 0, 1)

    def _generate_recommendations(self, results, clinical_data):

        Generate personalized treatment recommendations
        """
        recommendations = {

# EpiClock Integrated Analysis Platform
# Multi-stage build for optimized image size

# ============================================================================
# STAGE 1: Base Image with System Dependencies
# ============================================================================
FROM ubuntu:22.04 AS base

LABEL maintainer="Your Name <your.email@institution.edu>"
LABEL description="EpiClock: DNA Methylation + Variant Analysis Platform"
LABEL version="1.0.0"

# Prevent interactive prompts
ENV DEBIAN_FRONTEND=noninteractive
ENV TZ=Europe/Istanbul

# Install system dependencies
RUN apt-get update && apt-get install -y \
    # Build tools
    build-essential \
    cmake \
    git \
    wget \
    curl \
    unzip \
    # Python
    python3.10 \
    python3-pip \
    python3-dev \
    # R
    r-base \
    r-base-dev \
    # Bioinformatics tools
    bwa \
    samtools \
    bcftools \
    tabix \
    # Java (for GATK)
    openjdk-11-jdk \
    # VEP dependencies
    libdbi-perl \
    libdbd-mysql-perl \
    libarchive-zip-perl \
    # Utilities
    vim \
    htop \
    && rm -rf /var/lib/apt/lists/*

# Set working directory
WORKDIR /app

# ============================================================================
# STAGE 2: Python Environment
# ============================================================================
FROM base AS python-env

# Copy requirements
COPY requirements.txt /app/

# Install Python packages
RUN pip3 install --no-cache-dir --upgrade pip && \
    pip3 install --no-cache-dir -r requirements.txt

# Install additional bioinformatics Python packages
RUN pip3 install --no-cache-dir \
    pysam==0.21.0 \
    biopython==1.81 \
    cyvcf2==0.30.22 \
    pyvcf==0.6.8

# ============================================================================
# STAGE 3: R Environment
# ============================================================================
FROM python-env AS r-env

# Install R packages
RUN R -e "install.packages(c('BiocManager', 'devtools', 'remotes'), repos='https://cloud.r-project.org/')"

RUN R -e "BiocManager::install(c( \
    'minfi', \
    'limma', \
    'IlluminaHumanMethylation450kanno.ilmn12.hg19', \
    'IlluminaHumanMethylation450kmanifest', \
    'IlluminaHumanMethylationEPICanno.ilm10b4.hg19', \
    'IlluminaHumanMethylationEPICmanifest', \
    'GEOquery', \
    'sva', \
    'wateRmelon', \
    'ENmix' \
))"

RUN R -e "install.packages(c( \
    'ggplot2', \
    'dplyr', \
    'tidyr', \
    'data.table', \
    'R6', \
    'jsonlite' \
), repos='https://cloud.r-project.org/')"

# ============================================================================
# STAGE 4: Bioinformatics Tools
# ============================================================================
FROM r-env AS biotools

# Install GATK
RUN wget https://github.com/broadinstitute/gatk/releases/download/4.4.0.0/gatk-4.4.0.0.zip && \
    unzip gatk-4.4.0.0.zip && \
    mv gatk-4.4.0.0 /opt/gatk && \
    rm gatk-4.4.0.0.zip && \
    ln -s /opt/gatk/gatk /usr/local/bin/gatk

# Install VEP (Variant Effect Predictor)
RUN git clone https://github.com/Ensembl/ensembl-vep.git /opt/vep && \
    cd /opt/vep && \
    perl INSTALL.pl --AUTO a --SPECIES homo_sapiens --ASSEMBLY GRCh38 --NO_HTSLIB

ENV PATH="/opt/vep:${PATH}"

# Download reference genome (hg38)
RUN mkdir -p /data/reference && \
    cd /data/reference && \
    wget http://hgdownload.soe.ucsc.edu/goldenPath/hg38/bigZips/hg38.fa.gz && \
    gunzip hg38.fa.gz && \
    samtools faidx hg38.fa && \
    bwa index hg38.fa

# Download dbSNP
RUN cd /data/reference && \
    wget https://ftp.ncbi.nih.gov/snp/organisms/human_9606_b151_GRCh38p7/VCF/All_20180418.vcf.gz && \
    wget https://ftp.ncbi.nih.gov/snp/organisms/human_9606_b151_GRCh38p7/VCF/All_20180418.vcf.gz.tbi

# ============================================================================
# STAGE 5: EpiClock Application
# ============================================================================
FROM biotools AS epiclock

# Copy application code
COPY src/ /app/src/
COPY data/ /app/data/
COPY scripts/ /app/scripts/
COPY notebooks/ /app/notebooks/
COPY tests/ /app/tests/

# Install EpiClock package
COPY setup.py /app/
RUN pip3 install -e /app/

# Create output directory
RUN mkdir -p /app/output

# Set environment variables
ENV PYTHONPATH="/app/src:${PYTHONPATH}"
ENV R_LIBS_USER="/usr/local/lib/R/site-library"

# Expose Jupyter port
EXPOSE 8888

# Expose web API port
EXPOSE 5000

# Health check
HEALTHCHECK --interval=30s --timeout=10s --start-period=5s --retries=3 \
    CMD python3 -c "import epiclock; print('OK')" || exit 1

# Default command (Jupyter Lab)
CMD ["jupyter", "lab", "--ip=0.0.0.0", "--port=8888", "--no-browser", "--allow-root", "--NotebookApp.token=''"]

# Alternative commands:
# Run analysis: docker run epiclock python scripts/run_analysis.py
# Start API: docker run -p 5000:5000 epiclock python src/api/server.py
# Interactive shell: docker run -it epiclock bash

version: '3.8'

services:
  # ============================================================================
  # Main EpiClock Application
  # ============================================================================
  epiclock:
    build:
      context: .
      dockerfile: Dockerfile
    image: mortemdulcem/epiclock:latest
    container_name: epiclock-main
    ports:
      - "8888:8888"  # Jupyter Lab
      - "5000:5000"  # REST API
    volumes:
      # Mount data directories
      - ./data:/app/data
      - ./output:/app/output
      - ./user_data:/data/user_data
      # Mount notebooks for development
      - ./notebooks:/app/notebooks
    environment:
      - PYTHONUNBUFFERED=1
      - EPICLOCK_ENV=production
      - MAX_WORKERS=4
    command: jupyter lab --ip=0.0.0.0 --port=8888 --no-browser --allow-root --NotebookApp.token=''
    restart: unless-stopped
    networks:
      - epiclock-network

  # ============================================================================
  # REST API Server
  # ============================================================================
  api:
    build:
      context: .
      dockerfile: Dockerfile
    image: mortemdulcem/epiclock:latest
    container_name: epiclock-api
    ports:
      - "5001:5000"
    volumes:
      - ./data:/app/data
      - ./output:/app/output
    environment:
      - PYTHONUNBUFFERED=1
      - EPICLOCK_ENV=production
      - API_KEY=${API_KEY:-your-secret-key-here}
    command: python src/api/server.py
    restart: unless-stopped
    depends_on:
      - redis
    networks:
      - epiclock-network

  # ============================================================================
  # Redis (for job queue)
  # ============================================================================
  redis:
    image: redis:7-alpine
    container_name: epiclock-redis
    ports:
      - "6379:6379"
    volumes:
      - redis-data:/data
    restart: unless-stopped
    networks:
      - epiclock-network

  # ============================================================================
  # PostgreSQL (for results storage)
  # ============================================================================
  postgres:
    image: postgres:15-alpine
    container_name: epiclock-db
    ports:
      - "5432:5432"
    environment:
      - POSTGRES_DB=epiclock
      - POSTGRES_USER=epiclock
      - POSTGRES_PASSWORD=${DB_PASSWORD:-epiclock123}
    volumes:
      - postgres-data:/var/lib/postgresql/data
      - ./sql/init.sql:/docker-entrypoint-initdb.d/init.sql
    restart: unless-stopped
    networks:
      - epiclock-network

  # ============================================================================
  # Web Dashboard (Optional)
  # ============================================================================
  dashboard:
    build:
      context: ./web/frontend
      dockerfile: Dockerfile
    image: mortemdulcem/epiclock-dashboard:latest
    container_name: epiclock-dashboard
    ports:
      - "3000:3000"
    environment:
      - REACT_APP_API_URL=http://localhost:5001
    depends_on:
      - api
    restart: unless-stopped
    networks:
      - epiclock-network

  # ============================================================================
  # Nginx Reverse Proxy
  # ============================================================================

    image: nginx:alpine
    container_name: epiclock-nginx
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
      - ./nginx/ssl:/etc/nginx/ssl:ro
    depends_on:
      - epiclock
      - api
      - dashboard
    restart: unless-stopped
    networks:
      - epiclock-network

volumes:
  redis-data:

.PHONY: help build run stop clean test

# Default target
.DEFAULT_GOAL := help

# Variables
IMAGE_NAME = mortemdulcem/epiclock
VERSION = 1.0.0

help: ## Show this help message
	@echo "EpiClock Docker Management"
	@echo "=========================="
	@grep -E '^[a-zA-Z_-]+:.*?## .*$$' $(MAKEFILE_LIST) | awk 'BEGIN {FS = ":.*?## "}; {printf "\033[36m%-20s\033[0m %s\n", $$1, $$2}'

build: ## Build Docker image
	@echo "Building EpiClock Docker image..."
	docker build -t $(IMAGE_NAME):$(VERSION) -t $(IMAGE_NAME):latest .

build-no-cache: ## Build Docker image without cache
	@echo "Building EpiClock Docker image (no cache)..."
	docker build --no-cache -t $(IMAGE_NAME):$(VERSION) -t $(IMAGE_NAME):latest .

run: ## Run EpiClock container (Jupyter Lab)
	@echo "Starting EpiClock container..."
	docker-compose up -d epiclock
	@echo "Jupyter Lab: http://localhost:8888"

run-api: ## Run API server
	@echo "Starting EpiClock API server..."
	docker-compose up -d api
	@echo "API: http://localhost:5001"

run-all: ## Run all services
	@echo "Starting all EpiClock services..."
	docker-compose up -d
	@echo "Jupyter Lab: http://localhost:8888"
	@echo "API: http://localhost:5001"
	@echo "Dashboard: http://localhost:3000"

run-dev: ## Run development environment
	@echo "Starting development environment..."
	docker-compose -f docker-compose.dev.yml up

stop: ## Stop all containers
	@echo "Stopping EpiClock containers..."
	docker-compose down

stop-all: ## Stop and remove all containers, volumes
	@echo "Stopping and removing all EpiClock resources..."
	docker-compose down -v

logs: ## Show logs
	docker-compose logs -f

logs-api: ## Show API logs
	docker-compose logs -f api

shell: ## Open shell in container
	docker-compose exec epiclock bash

shell-api: ## Open shell in API container
	docker-compose exec api bash

test: ## Run tests
	docker-compose exec epiclock pytest tests/ -v

test-coverage: ## Run tests with coverage
	docker-compose exec epiclock pytest tests/ --cov=src --cov-report=html

clean: ## Clean up Docker resources
	@echo "Cleaning up Docker resources..."
	docker-compose down -v
	docker system prune -f

push: ## Push image to Docker Hub
	@echo "Pushing image to Docker Hub..."
	docker push $(IMAGE_NAME):$(VERSION)
	docker push $(IMAGE_NAME):latest

pull: ## Pull image from Docker Hub
	@echo "Pulling image from Docker Hub..."
	docker pull $(IMAGE_NAME):latest

# Quick analysis commands
analyze-example: ## Run analysis on example data
	docker-compose exec epiclock python scripts/run_analysis.py \
		--methylation data/example/synthetic_100samples.csv \
		--vcf data/example/synthetic_variants.vcf \
		--output output/example_results

analyze-geo: ## Download and analyze GEO dataset
	docker-compose exec epiclock python scripts/run_analysis.py \
		--geo GSE110043 \
		--output output/geo_results

# Utility commands
update-deps: ## Update Python dependencies
	docker-compose exec epiclock pip install --upgrade -r requirements.txt

jupyter-token: ## Get Jupyter token
	docker-compose exec epiclock jupyter lab list

stats: ## Show container stats
	docker stats epiclock-main epiclock-api

backup: ## Backup data and results
	@echo "Creating backup..."
	tar -czf epiclock-backup-$(shell date +%Y%m%d-%H%M%S).tar.gz data/ output/

restore: ## Restore from backup (usage: make restore BACKUP=file.tar.gz)
	@echo "Restoring from $(BACKUP)..."
	tar -xzf $(BACKUP)
