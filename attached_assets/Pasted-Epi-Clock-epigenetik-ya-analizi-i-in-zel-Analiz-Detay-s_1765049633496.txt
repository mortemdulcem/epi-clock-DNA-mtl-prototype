    // Sekme geçişi
    const tabLinks = document.querySelectorAll('.tab-link');
    const tabContents = {
      overview: document.getElementById('tab-overview'),
      data: document.getElementById('tab-data'),
      model: document.getElementById('tab-model'),
      predictions: document.getElementById('tab-predictions'),
      explainability: document.getElementById('tab-explainability')
    };

    tabLinks.forEach(btn => {
      btn.addEventListener('click', () => {
        const target = btn.getAttribute('data-tab');

        tabLinks.forEach(b => {
          b.classList.remove('border-unblue', 'text-unblue', 'border-b-2');
          b.classList.add('text-slate-600');
        });
        btn.classList.remove('text-slate-600');
        btn.classList.add('border-b-2', 'border-unblue', 'text-unblue');

        Object.keys(tabContents).forEach(key => {
          tabContents[key].classList.remove('active');
        });
        tabContents[target].classList.add('active');
      });
    });

    // Epi-Clock API DEMO entegrasyonu (backend hazır olunca gerçek endpoint ile değiştir)
    const runBtn = document.getElementById('run-epiclock-btn');
    const epiclockResultDiv = document.getElementById('epiclock-result');

    runBtn.addEventListener('click', async () => {
      const sampleId = 'SAMPLE-001';

      try {
        const res = await fetch('/api/epiclock/predict', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ sample_id: sampleId })
        });

        if (!res.ok) throw new Error('API hatası');

        const data = await res.json();

        epiclockResultDiv.innerHTML = `
          <p class="text-xs text-slate-700"><span class="font-semibold">Örnek ID:</span> ${data.sample_id}</p>
          <p class="text-xs text-slate-700"><span class="font-semibold">Epigenetik yaş (tahmin):</span> ${data.predicted_epigenetic_age.toFixed(1)} yıl</p>
          <p class="text-xs text-slate-700"><span class="font-semibold">Yaş ivmesi:</span> ${data.age_acceleration.toFixed(1)} yıl</p>
          <p class="text-xs text-slate-700">
            <span class="font-semibold">MAE:</span> ${data.mae.toFixed(2)} ·
            <span class="font-semibold">RMSE:</span> ${data.rmse.toFixed(2)}
          </p>
        `;
      } catch (err) {
        console.error(err);
        epiclockResultDiv.innerHTML = `
          <p class="text-xs text-red-600">Epi-Clock analizi çalıştırılırken bir hata oluştu veya API henüz bağlı değil.</p>
        `;
      }
    });
  </script>
