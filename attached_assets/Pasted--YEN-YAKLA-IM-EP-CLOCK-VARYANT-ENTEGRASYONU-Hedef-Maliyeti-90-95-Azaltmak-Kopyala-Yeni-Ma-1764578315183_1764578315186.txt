"""
Variant Calling Module
Integrates GATK best practices with EpiClock pipeline
"""

import subprocess
import pandas as pd
from pathlib import Path

class VariantCaller:
    """
    Automated variant calling from sequencing data.
    
    Supports:
    - Illumina short reads (FASTQ)
    - BAM/CRAM alignment files
    - VCF output
    
    Pipeline:
    1. Quality control (FastQC)
    2. Alignment (BWA-MEM)
    3. Variant calling (GATK HaplotypeCaller)
    4. Filtering (GATK VQSR)
    """
    
    def __init__(self, reference_genome="hg38", threads=8):
        self.reference = reference_genome
        self.threads = threads
        self.gatk_path = "gatk"  # Assumes GATK in PATH
        
    def call_variants(self, fastq_r1, fastq_r2, output_vcf):
        """
        Full pipeline: FASTQ → VCF
        
        Args:
            fastq_r1: Forward reads
            fastq_r2: Reverse reads
            output_vcf: Output VCF file
            
        Returns:
            Path to filtered VCF
        """
        # Step 1: Alignment
        bam = self._align_reads(fastq_r1, fastq_r2)
        
        # Step 2: Mark duplicates
        dedup_bam = self._mark_duplicates(bam)
        
        # Step 3: Base recalibration
        recal_bam = self._base_recalibration(dedup_bam)
        
        # Step 4: Variant calling
        raw_vcf = self._haplotype_caller(recal_bam)
        
        # Step 5: Filtering
        filtered_vcf = self._filter_variants(raw_vcf, output_vcf)
        
        return filtered_vcf
    
    def _align_reads(self, r1, r2):
        """BWA-MEM alignment"""
        cmd = f"""
        bwa mem -t {self.threads} \
            {self.reference} {r1} {r2} | \
        samtools sort -@ {self.threads} -o aligned.bam -
        samtools index aligned.bam
        """
        subprocess.run(cmd, shell=True, check=True)
        return "aligned.bam"
    
    def _mark_duplicates(self, bam):
        """GATK MarkDuplicates"""
        cmd = f"""
        {self.gatk_path} MarkDuplicates \
            -I {bam} \
            -O dedup.bam \
            -M metrics.txt
        samtools index dedup.bam
        """
        subprocess.run(cmd, shell=True, check=True)
        return "dedup.bam"
    
    def _base_recalibration(self, bam):
        """GATK BQSR"""
        cmd = f"""
        {self.gatk_path} BaseRecalibrator \
            -I {bam} \
            -R {self.reference} \
            --known-sites dbsnp.vcf \
            -O recal_data.table
        
        {self.gatk_path} ApplyBQSR \
            -I {bam} \
            -R {self.reference} \
            --bqsr-recal-file recal_data.table \
            -O recal.bam
        """
        subprocess.run(cmd, shell=True, check=True)
        return "recal.bam"
    
    def _haplotype_caller(self, bam):
        """GATK HaplotypeCaller"""
        cmd = f"""
        {self.gatk_path} HaplotypeCaller \
            -I {bam} \
            -R {self.reference} \
            -O raw_variants.vcf \
            --native-pair-hmm-threads {self.threads}
        """
        subprocess.run(cmd, shell=True, check=True)
        return "raw_variants.vcf"
    
    def _filter_variants(self, vcf, output):
        """GATK VariantFiltration"""
        cmd = f"""
        {self.gatk_path} VariantFiltration \
            -V {vcf} \
            -O {output} \
            --filter-expression "QD < 2.0" --filter-name "QD2" \
            --filter-expression "FS > 60.0" --filter-name "FS60" \
            --filter-expression "MQ < 40.0" --filter-name "MQ40"
        """
        subprocess.run(cmd, shell=True, check=True)
        return output

"""
Variant Annotation Module
Functional annotation and clinical interpretation
"""

import subprocess
import pandas as pd
from typing import List, Dict

class VariantAnnotator:
    """
    Annotate variants with functional and clinical information.
    
    Uses:
    - VEP (Variant Effect Predictor)
    - ClinVar
    - gnomAD
    - PharmGKB
    """
    
    def __init__(self, vep_cache_dir="/data/vep_cache"):
        self.vep_cache = vep_cache_dir
        
    def annotate(self, vcf_file, output_file):
        """
        Full annotation pipeline
        
        Args:
            vcf_file: Input VCF
            output_file: Annotated output (TSV)
            
        Returns:
            DataFrame with annotations
        """
        # VEP annotation
        vep_output = self._run_vep(vcf_file)
        
        # Parse VEP output
        df = self._parse_vep(vep_output)
        
        # Add clinical annotations
        df = self._add_clinvar(df)
        df = self._add_pharmgkb(df)
        
        # Filter to functional variants
        df = self._filter_functional(df)
        
        # Save
        df.to_csv(output_file, sep='\t', index=False)
        return df
    
    def _run_vep(self, vcf):
        """Run Ensembl VEP"""
        cmd = f"""
        vep --input_file {vcf} \
            --output_file vep_output.txt \
            --cache --dir_cache {self.vep_cache} \
            --assembly GRCh38 \
            --format vcf \
            --tab \
            --fields "Uploaded_variation,Location,Allele,Gene,SYMBOL,Consequence,IMPACT,SIFT,PolyPhen,gnomAD_AF" \
            --fork 4
        """
        subprocess.run(cmd, shell=True, check=True)
        return "vep_output.txt"
    
    def _parse_vep(self, vep_file):
        """Parse VEP output to DataFrame"""
        df = pd.read_csv(vep_file, sep='\t', comment='#')
        return df
    
    def _add_clinvar(self, df):
        """Add ClinVar clinical significance"""
        # Load ClinVar database
        clinvar = pd.read_csv("clinvar.tsv", sep='\t')
        
        # Merge
        df = df.merge(
            clinvar[['variant_id', 'clinical_significance']],
            left_on='Uploaded_variation',
            right_on='variant_id',
            how='left'
        )
        return df
    
    def _add_pharmgkb(self, df):
        """Add PharmGKB drug-gene interactions"""
        pharmgkb = pd.read_csv("pharmgkb.tsv", sep='\t')
        
        df = df.merge(
            pharmgkb[['gene', 'drug', 'phenotype']],
            left_on='SYMBOL',
            right_on='gene',
            how='left'
        )
        return df
    
    def _filter_functional(self, df):
        """Filter to likely functional variants"""
        functional_consequences = [
            'missense_variant',
            'stop_gained',
            'frameshift_variant',
            'splice_donor_variant',
            'splice_acceptor_variant'
        ]
        
        df_filtered = df[
            df['Consequence'].isin(functional_consequences) |
            (df['IMPACT'] == 'HIGH') |
            (df['clinical_significance'].str.contains('Pathogenic', na=False))
        ]
        
        return df_filtered

"""
Polygenic Risk Score (PRS) Module
Calculate genetic risk for substance use disorders
"""

import pandas as pd
import numpy as np
from typing import Dict, List, Tuple
from pathlib import Path

class PolygenicRiskCalculator:
    """
    Calculate Polygenic Risk Scores for addiction-related traits.
    
    Traits:
    - Alcohol dependence
    - Opioid dependence
    - Cocaine dependence
    - Nicotine dependence
    - Cannabis use disorder
    - General addiction liability
    
    Uses GWAS summary statistics from:
    - Psychiatric Genomics Consortium (PGC)
    - GWAS Catalog
    - UK Biobank
    """
    
    def __init__(self, gwas_dir="/data/gwas_sumstats"):
        self.gwas_dir = Path(gwas_dir)
        self.prs_weights = self._load_prs_weights()
        
    def calculate_prs(self, variants_df, trait='alcohol_dependence'):
        """
        Calculate PRS for a specific trait
        
        Args:
            variants_df: DataFrame with individual's variants
            trait: Trait name (alcohol_dependence, opioid_dependence, etc.)
            
        Returns:
            Dictionary with PRS and percentile
        """
        # Load trait-specific weights
        weights = self.prs_weights[trait]
        
        # Match variants
        matched = self._match_variants(variants_df, weights)
        
        # Calculate raw PRS
        raw_prs = self._calculate_raw_prs(matched)
        
        # Standardize (z-score)
        standardized_prs = self._standardize_prs(raw_prs, trait)
        
        # Convert to percentile
        percentile = self._prs_to_percentile(standardized_prs)
        
        # Interpret risk
        risk_category = self._interpret_risk(percentile)
        
        return {
            'trait': trait,
            'raw_prs': raw_prs,
            'standardized_prs': standardized_prs,
            'percentile': percentile,
            'risk_category': risk_category,
            'n_variants_used': len(matched),
            'interpretation': self._generate_interpretation(trait, percentile, risk_category)
        }
    
    def calculate_multi_trait_prs(self, variants_df):
        """
        Calculate PRS for all addiction-related traits
        
        Args:
            variants_df: DataFrame with individual's variants
            
        Returns:
            Dictionary with PRS for each trait
        """
        traits = [
            'alcohol_dependence',
            'opioid_dependence',
            'cocaine_dependence',
            'nicotine_dependence',
            'cannabis_use_disorder',
            'general_addiction_liability'
        ]
        
        results = {}
        for trait in traits:
            results[trait] = self.calculate_prs(variants_df, trait)
        
        # Calculate composite risk
        results['composite'] = self._calculate_composite_risk(results)
        
        return results
    
    def _load_prs_weights(self):
        """
        Load PRS weights from GWAS summary statistics
        
        Format:
        SNP | CHR | BP | A1 | A2 | BETA | P
        """

    def _match_variants(self, variants_df, weights_df):
        """
        Match individual's variants with PRS weights
        
        Returns:
            DataFrame with matched variants and their weights
        """
        # Merge on SNP ID or position
        matched = variants_df.merge(
            weights_df,
            left_on='Uploaded_variation',
            right_on='SNP',
            how='inner'
        )
        
        # Extract dosage (0, 1, 2 copies of effect allele)
        matched['dosage'] = matched.apply(self._calculate_dosage, axis=1)
        
        return matched
    
    def _calculate_dosage(self, row):
        """
        Calculate allele dosage (0, 1, or 2)
        
        Genotype encoding:
        - 0/0 (ref/ref): 0 copies
        - 0/1 (ref/alt): 1 copy
        - 1/1 (alt/alt): 2 copies
        """
        # Parse genotype from VCF
        genotype = row.get('genotype', '0/0')
        
        if genotype == '0/0':
            return 0
        elif genotype in ['0/1', '1/0']:
            return 1
        elif genotype == '1/1':
            return 2
        else:
            return np.nan
    
    def _calculate_raw_prs(self, matched_df):
        """
        Calculate raw PRS
        
        PRS = Σ(beta_i × dosage_i)
        """
        prs = (matched_df['BETA'] * matched_df['dosage']).sum()
        return prs
    
    def _standardize_prs(self, raw_prs, trait):

        Standardize PRS to z-score
        
        Uses population mean and SD from reference panel

        # Load reference statistics

    def _analyze_epigenetic_age(self, methylation_data, clinical_data):
        """
        Analyze epigenetic age acceleration
        """
        # Calculate all clocks
        ages = self.clock_calculator.calculate_all_clocks(methylation_data)
        
        # Calculate acceleration
        chronological_age = clinical_data['age']
        acceleration = {}
        
        for clock, epi_age in ages.items():
            acceleration[clock] = epi_age - chronological_age
        
        # Clinical interpretation
        interpretation = self.clinical_support.interpret_acceleration(
            acceleration,
            clinical_data
        )
        
        return {
            'chronological_age': chronological_age,
            'epigenetic_ages': ages,
            'acceleration': acceleration,
            'mean_acceleration': np.mean(list(acceleration.values())),
            'interpretation': interpretation
        }
    
    def _calculate_integrated_risk(self, results, clinical_data):
        """
        Calculate integrated risk score combining all data sources
        
        Risk factors:
        1. Epigenetic age acceleration (weight: 0.3)
        2. Polygenic risk score (weight: 0.3)
        3. Pharmacogenomic risk variants (weight: 0.2)
        4. Clinical history (weight: 0.2)
        """
        # 1. Epigenetic component
        epi_acceleration = results['epigenetic']['mean_acceleration']
        epi_risk = self._normalize_risk(epi_acceleration, min_val=-5, max_val=10)
        
        # 2. Genetic component (PRS)
        composite_prs = results['prs']['composite']['percentile']
        prs_risk = composite_prs / 100  # Convert to 0-1 scale
        
        # 3. Pharmacogenomic component
        pgx_risk = self._assess_pgx_risk(results['pharmacogenomics'])
        
        # 4. Clinical component
        clinical_risk = self._assess_clinical_risk(clinical_data)
        
        # Weighted combination
        integrated_risk_score = (
            0.3 * epi_risk +
            0.3 * prs_risk +
            0.2 * pgx_risk +
            0.2 * clinical_risk
        )
        
        # Convert to percentile
        risk_percentile = integrated_risk_score * 100
        
        # Categorize
        if risk_percentile < 25:
            risk_category = "Low"
        elif risk_percentile < 50:
            risk_category = "Moderate"
        elif risk_percentile < 75:
            risk_category = "High"
        else:
            risk_category = "Very High"
        
        return {
            'integrated_risk_score': integrated_risk_score,
            'risk_percentile': risk_percentile,
            'risk_category': risk_category,
            'components': {
                'epigenetic': epi_risk,
                'genetic_prs': prs_risk,
                'pharmacogenomic': pgx_risk,
                'clinical': clinical_risk
            },
            'interpretation': self._interpret_integrated_risk(
                risk_percentile, 
                risk_category,
                results
            )
        }
    
    def _normalize_risk(self, value, min_val, max_val):
        """Normalize value to 0-1 scale"""
        normalized = (value - min_val) / (max_val - min_val)
        return np.clip(normalized, 0, 1)
    
    def _assess_pgx_risk(self, pgx_results):
        """
        Assess pharmacogenomic risk
        
        High-risk factors:
        - CYP2D6 poor metabolizer
        - DRD2 Taq1A A1/A1 (high addiction risk)
        - COMT Val/Val (poor treatment response)
        """
        risk_score = 0.5  # Baseline
        
        # Opioid metabolism
        if pgx_results['opioid']['cyp2d6_phenotype'] == "Poor Metabolizer (PM)":
            risk_score += 0.2
        elif pgx_results['opioid']['cyp2d6_phenotype'] == "Ultra-rapid Metabolizer (UM)":
            risk_score += 0.15
        
        # Addiction risk (DRD2)
        addiction_risk = pgx_results['stimulant']['addiction_risk']
        if addiction_risk == "INCREASED":
            risk_score += 0.2
        elif addiction_risk == "MODERATE":
            risk_score += 0.1
        
        # Alcohol protection (ALDH2)
        alcohol_risk = pgx_results['alcohol']['alcohol_dependence_risk']
        if alcohol_risk == "LOW":
            risk_score -= 0.2
        
        return np.clip(risk_score, 0, 1)
    
    def _assess_clinical_risk(self, clinical_data):
        """
        Assess clinical risk based on history
        
        Risk factors:
        - Substance use duration
        - Number of substances
        - Family history
        - Age of onset
        - Previous treatment failures
        """
        risk_score = 0.0
        
        # Duration of use
        if 'substance_use_years' in clinical_data:
            years = clinical_data['substance_use_years']
            if years > 10:
                risk_score += 0.3
            elif years > 5:
                risk_score += 0.2
            elif years > 2:
                risk_score += 0.1
        
        # Polysubstance use
        if 'n_substances' in clinical_data:
            n_sub = clinical_data['n_substances']
            if n_sub >= 3:
                risk_score += 0.3
            elif n_sub == 2:
                risk_score += 0.2
        
        # Family history
        if clinical_data.get('family_history_addiction', False):
            risk_score += 0.2
        
        # Early onset
        if 'age_of_onset' in clinical_data:
            if clinical_data['age_of_onset'] < 15:
                risk_score += 0.3
            elif clinical_data['age_of_onset'] < 18:
                risk_score += 0.2
        
        # Treatment failures
        if 'n_treatment_failures' in clinical_data:
            failures = clinical_data['n_treatment_failures']
            risk_score += min(0.3, failures * 0.1)
        
        return np.clip(risk_score, 0, 1)
