class SubstanceDetector:
    """
    DNA metilasyon verilerinden madde kullanımı tespiti
    """
    
    def __init__(self, model_path='models/substance_classifier_v2.pkl'):
        """
        Modeli yükle
        """
        self.model_path = model_path
        self.load_models()
        
    def load_models(self):
        """
        Önceden eğitilmiş modelleri yükle
        """
        try:
            with open(self.model_path, 'rb') as f:
                models = pickle.load(f)
                
            self.substance_classifier = models['classifier']  # Random Forest
            self.duration_estimator = models['duration']  # Gradient Boosting
            self.timing_estimator = models['timing']  # Neural Network
            self.scaler = models['scaler']
            self.pca = models['pca']
            
            print("✓ Modeller başarıyla yüklendi")
            
        except FileNotFoundError:
            print("⚠ Model dosyası bulunamadı, yeni model eğitiliyor...")
            self.train_models()
    
    def preprocess_methylation_data(self, beta_values):
        """
        Metilasyon verisini ön işle
        
        Args:
            beta_values: DataFrame, 850K CpG × 1 sample
            
        Returns:
            processed_data: numpy array
        """
        # 1. Kalite kontrol
        beta_values = self.quality_control(beta_values)
        
        # 2. Normalizasyon (BMIQ)
        beta_values = self.bmiq_normalization(beta_values)
        
        # 3. Batch effect correction (ComBat)
        beta_values = self.combat_correction(beta_values)
        
        # 4. Madde-spesifik CpG'leri çıkar
        substance_cpgs = self.extract_substance_cpgs(beta_values)
        
        # 5. Standardize
        substance_cpgs_scaled = self.scaler.transform(substance_cpgs)
        
        # 6. PCA (boyut indirgeme)
        substance_cpgs_pca = self.pca.transform(substance_cpgs_scaled)
        
        return substance_cpgs_pca
    
    def quality_control(self, beta_values):
        """
        Kalite kontrol
        """
        # Detection p-value > 0.01 olan probe'ları çıkar
        # Çapraz-reaktif probe'ları çıkar
        # SNP içeren probe'ları çıkar
        # X, Y kromozom probe'larını çıkar
        
        return beta_values
    
    def bmiq_normalization(self, beta_values):
        """
        Beta-Mixture Quantile normalization
        """
        # Type I ve Type II probe'ları ayır
        # Quantile normalization uygula
        
        return beta_values
    
    def combat_correction(self, beta_values):
        """
        Batch effect düzeltme
        """
        # ComBat algoritması
        
        return beta_values
    
    def extract_substance_cpgs(self, beta_values):
        """
        Madde-spesifik CpG sitelerini çıkar
        """
        all_substance_cpgs = []
        
        for substance_class, data in CPG_SIGNATURES.items():
            acute_cpgs = [sig['cpg'] for sig in data['acute']]
            chronic_cpgs = [sig['cpg'] for sig in data['chronic']]
            all_substance_cpgs.extend(acute_cpgs + chronic_cpgs)
        
        # Tekrarları kaldır
        unique_cpgs = list(set(all_substance_cpgs))
        
        # Beta değerlerini çıkar
        substance_beta = beta_values.loc[unique_cpgs]
        
        return substance_beta.values.reshape(1, -1)
    
    def detect_substance(self, beta_values):
        """
        Madde tespiti yap
        
        Args:
            beta_values: DataFrame, metilasyon verileri
            
        Returns:
            results: dict, tespit sonuçları
        """
        # Ön işleme
        processed_data = self.preprocess_methylation_data(beta_values)
        
        # 1. Madde sınıfı tahmini
        substance_probs = self.substance_classifier.predict_proba(processed_data)[0]
        substance_classes = self.substance_classifier.classes_
        
        # En yüksek olasılıklı 5 madde
        top_5_indices = np.argsort(substance_probs)[-5:][::-1]
        top_5_substances = [
            {
                'substance_class': substance_classes[i],
                'probability': float(substance_probs[i]),
                'confidence': self.calculate_confidence(substance_probs[i])
            }
            for i in top_5_indices
        ]
        
        # 2. Kullanım süresi tahmini
        predicted_substance = substance_classes[top_5_indices[0]]
        duration_days = self.duration_estimator.predict(processed_data)[0]
        
        # 3. Son kullanım zamanı tahmini
        last_use_days = self.timing_estimator.predict(processed_data)[0]
        
        # 4. Kronik/akut ayrımı
        usage_pattern = self.classify_usage_pattern(duration_days, last_use_days)
        
        # 5. Detaylı CpG analizi
        cpg_analysis = self.analyze_cpg_signatures(beta_values, predicted_substance)
        
        # Sonuçları derle
        results = {
            'detected_substances': top_5_substances,
            'primary_substance': {
                'class': predicted_substance,
                'probability': float(substance_probs[top_5_indices[0]]),
                'specific_compounds': self.identify_specific_compounds(
                    beta_values, predicted_substance
                )
            },
            'usage_duration': {
                'estimated_days': float(duration_days),
                'estimated_months': float(duration_days / 30),
                'estimated_years': float(duration_days / 365),
                'confidence_interval_95': self.calculate_duration_ci(duration_days)
            },
            'last_use': {
                'estimated_days_ago': float(last_use_days),
                'estimated_date': (datetime.now() - timedelta(days=last_use_days)).strftime('%Y-%m-%d'),
                'confidence_interval_95': self.calculate_timing_ci(last_use_days)
            },
            'usage_pattern': usage_pattern,
            'cpg_signature_analysis': cpg_analysis,
            'epigenetic_age_acceleration': self.calculate_eaa(beta_values),
            'health_risk_assessment': self.assess_health_risks(
                predicted_substance, duration_days, beta_values
            ),
            'recovery_timeline': self.estimate_recovery(
                predicted_substance, duration_days, last_use_days
            )
        }
        
        return results
    
    def calculate_confidence(self, probability):
        """
        Güven seviyesi hesapla
        """
        if probability > 0.9:
            return 'VERY HIGH'
        elif probability > 0.75:
            return 'HIGH'
        elif probability > 0.6:
            return 'MODERATE'
        elif probability > 0.4:
            return 'LOW'
        else:
            return 'VERY LOW'
    
    def classify_usage_pattern(self, duration_days, last_use_days):
        """
        Kullanım paternini sınıflandır
        """
        if duration_days < 30:
            if last_use_days < 7:
                return {
                    'pattern': 'ACUTE_RECENT',
                    'description': 'Akut kullanım, yakın zamanda',
                    'severity': 'MILD'
                }
            else:
                return {
                    'pattern': 'ACUTE_PAST',
                    'description': 'Akut kullanım, geçmişte',
                    'severity': 'MILD'
                }
        elif duration_days < 180:
            return {
                'pattern': 'SUBCHRONIC',
                'description': 'Subkronik kullanım (1-6 ay)',
                'severity': 'MODERATE'
            }
        else:
            if last_use_days < 30:
                return {
                    'pattern': 'CHRONIC_ACTIVE',
                    'description': 'Kronik kullanım, aktif',
                    'severity': 'SEVERE'
                }
            else:
                return {
                    'pattern': 'CHRONIC_RECOVERY',
                    'description': 'Kronik kullanım, iyileşme döneminde',
                    'severity': 'MODERATE'
                }
    
    def analyze_cpg_signatures(self, beta_values, substance_class):
        """
        CpG imza analizi
        """
        if substance_class not in CPG_SIGNATURES:
            return None
        
        signatures = CPG_SIGNATURES[substance_class]
        
        analysis = {
            'acute_markers': [],
            'chronic_markers': [],
            'overall_match': 0.0
        }
        
        # Akut markerlar
        for sig in signatures['acute']:
            cpg = sig['cpg']
            if cpg in beta_values.index:
                observed_beta = beta_values.loc[cpg].values[0]
                expected_direction = sig['direction']
                expected_delta = sig['delta_beta']
                
                # Kontrol grubu ortalaması (varsayılan: 0.5)
                control_mean = 0.5
                observed_delta = observed_beta - control_mean
                
                # Yön kontrolü
                direction_match = (
                    (expected_direction == 'hyper' and observed_delta > 0) or
                    (expected_direction == 'hypo' and observed_delta < 0)
                )
                
                # Büyüklük kontrolü
                magnitude_match = abs(observed_delta) >= abs(expected_delta) * 0.5
                
                analysis['acute_markers'].append({
                    'cpg': cpg,
                    'gene': sig['gene'],
                    'expected_delta': expected_delta,
                    'observed_delta': float(observed_delta),
                    'direction_match': direction_match,
                    'magnitude_match': magnitude_match,
                    'confidence': float(abs(observed_delta) / abs(expected_delta)) if expected_delta != 0 else 0
                })
        
        # Kronik markerlar
        for sig in signatures['chronic']:
            cpg = sig['cpg']
            if cpg in beta_values.index:
                observed_beta = beta_values.loc[cpg].values[0]
                expected_direction = sig['direction']
                expected_delta = sig['delta_beta']
                
                control_mean = 0.5
                observed_delta = observed_beta - control_mean
                
                direction_match = (
                    (expected_direction == 'hyper' and observed_delta > 0) or
                    (expected_direction == 'hypo' and observed_delta < 0)
                )
                
                magnitude_match = abs(observed_delta) >= abs(expected_delta) * 0.5
                
                analysis['chronic_markers'].append({
                    'cpg': cpg,
                    'gene': sig['gene'],
                    'expected_delta': expected_delta,
                    'observed_delta': float(observed_delta),
                    'direction_match': direction_match,
                    'magnitude_match': magnitude_match,
                    'confidence': float(abs(observed_delta) / abs(expected_delta)) if expected_delta != 0 else 0
                })
        
        # Genel eşleşme skoru
        all_markers = analysis['acute_markers'] + analysis['chronic_markers']
        if all_markers:
            direction_matches = sum(1 for m in all_markers if m['direction_match'])
            magnitude_matches = sum(1 for m in all_markers if m['magnitude_match'])
            analysis['overall_match'] = (direction_matches + magnitude_matches) / (2 * len(all_markers))
        
        return analysis
    
    def identify_specific_compounds(self, beta_values, substance_class):
        """
        Spesifik bileşikleri tanımla
        """
        if substance_class not in SUBSTANCE_DATABASE:
            return []
        
        db = SUBSTANCE_DATABASE[substance_class]
        specific_compounds = []
        
        # Her alt kategori için
        for category, compounds in db.items():
            if category in ['cpg_signature', 'half_life_days']:
                continue
            
            if isinstance(compounds, list):
                # İlk 10 bileşik (örnek)
                specific_compounds.extend(compounds[:10])
        
        return specific_compounds
    
    def calculate_eaa(self, beta_values):
        """
        Epigenetik yaş ivmelenmesi hesapla
        """
        # Horvath saati kullan
        from epiclock.clocks import EpigeneticClockCalculator
        
        calculator = EpigeneticClockCalculator()
        horvath_age = calculator.calculate_age(beta_values, clock='horvath')
        
        # Kronolojik yaş (klinik veriden alınmalı, burada varsayılan)
        chronological_age = 35  # Placeholder
        
        eaa = horvath_age - chronological_age
        
        return {
            'epigenetic_age': float(horvath_age),
            'acceleration_years': float(eaa),
            'interpretation': self.interpret_eaa(eaa)
        }
    
    def interpret_eaa(self, eaa):
        """
        EAA yorumla
        """
        if eaa > 5:
            return 'Belirgin epigenetik yaşlanma (>5 yıl), yüksek sağlık riski'
        elif eaa > 2:
            return 'Orta düzey epigenetik yaşlanma (2-5 yıl), artmış risk'
        elif eaa > -2:
            return 'Normal epigenetik yaş'
        else:
            return 'Yavaş epigenetik yaşlanma (<-2 yıl)'
    
    def assess_health_risks(self, substance_class, duration_days, beta_values):
        """
        Sağlık riski değerlendirmesi
        """
        risks = {
            'cardiovascular': 0,
            'hepatic': 0,
            'neurological': 0,
            'psychiatric': 0,
            'metabolic': 0,
            'overall': 0
        }
        
        # Madde-spesifik risk faktörleri
        risk_factors = {
            'alcohol': {
                'hepatic': 0.8,
                'cardiovascular': 0.6,
                'neurological': 0.5,
                'psychiatric': 0.4,
                'metabolic': 0.7
            },
            'opioids': {
                'neurological': 0.7,
                'psychiatric': 0.6,
                'cardiovascular': 0.3,
                'hepatic': 0.2,
                'metabolic': 0.3
            },
            'stimulants': {
                'cardiovascular': 0.9,
                'psychiatric': 0.8,
                'neurological': 0.6,
                'metabolic': 0.5,
                'hepatic': 0.3
            },
            'cannabis': {
                'psychiatric': 0.6,
                'neurological': 0.4,
                'cardiovascular': 0.2,
                'hepatic': 0.1,
                'metabolic': 0.2
            }
        }
        
        if substance_class in risk_factors:
            base_risks = risk_factors[substance_class]
            
            # Süre faktörü (logaritmik)
            duration_factor = min(np.log10(duration_days + 1) / 3, 1.0)
            
            for organ, base_risk in base_risks.items():
                risks[organ] = base_risk * duration_factor
        
        # Genel risk
        risks['overall'] = np.mean(list(risks.values()))
        
        # Risk kategorisi
        overall_risk = risks['overall']
        if overall_risk > 0.7:
            risk_category = 'VERY HIGH'
        elif overall_risk > 0.5:
            risk_category = 'HIGH'
        elif overall_risk > 0.3:
            risk_category = 'MODERATE'
        else:
            risk_category = 'LOW'
        
        return {
            'risk_scores': risks,
            'risk_category': risk_category,
            'recommendations': self.generate_recommendations(risks, substance_class)
        }
    
    def generate_recommendations(self, risks, substance_class):
        """
        Öneriler oluştur
        """
        recommendations = []
        
        if risks['cardiovascular'] > 0.5:
            recommendations.append('Kardiyoloji konsültasyonu önerilir')
            recommendations.append('EKG ve ekokardiyografi yapılmalı')
        
        if risks['hepatic'] > 0.5:
            recommendations.append('Karaciğer fonksiyon testleri (AST, ALT, GGT)')
            recommendations.append('Hepatoloji değerlendirmesi')
        
        if risks['neurological'] > 0.5:
            recommendations.append('Nörolojik muayene')
            recommendations.append('Beyin MRI düşünülmeli')
        
        if risks['psychiatric'] > 0.5:
            recommendations.append('Psikiyatri konsültasyonu acil')
            recommendations.append('Bağımlılık tedavi programı')
        
        if risks['metabolic'] > 0.5:
            recommendations.append('Metabolik panel (glukoz, lipidler, HbA1c)')
            recommendations.append('Endokrinoloji değerlendirmesi')
        
        return recommendations
    
    def estimate_recovery(self, substance_class, duration_days, last_use_days):
        """
        İyileşme süresi tahmini
        """
        if substance_class not in CPG_SIGNATURES:
            return None
        
        time_decay = CPG_SIGNATURES[substance_class]['time_decay']
        half_life = time_decay['half_life_days']
        full_recovery = time_decay['recovery_days']
        
        # Metilasyon düzelmesi (eksponansiyel decay)
        current_recovery = 1 - np.exp(-last_use_days / half_life)
        remaining_days = full_recovery - last_use_days
        
        return {
            'methylation_half_life_days': half_life,
            'full_recovery_days': full_recovery,
            'current_recovery_percent': float(current_recovery * 100),
            'remaining_recovery_days': max(0, float(remaining_days)),
            'estimated_full_recovery_date': (
                datetime.now() + timedelta(days=max(0, remaining_days))
            ).strftime('%Y-%m-%d'),
            'recovery_milestones': [
                {
                    'milestone': '25% düzelme',
                    'days': half_life * 0.42,
                    'date': (datetime.now() + timedelta(days=half_life * 0.42 - last_use_days)).strftime('%Y-%m-%d') if last_use_days < half_life * 0.42 else 'Geçildi'
                },
                {
                    'milestone': '50% düzelme',
                    'days': half_life,
                    'date': (datetime.now() + timedelta(days=half_life - last_use_days)).strftime('%Y-%m-%d') if last_use_days < half_life else 'Geçildi'
                },
                {
                    'milestone': '75% düzelme',
                    'days': half_life * 2,
                    'date': (datetime.now() + timedelta(days=half_life * 2 - last_use_days)).strftime('%Y-%m-%d') if last_use_days < half_life * 2 else 'Geçildi'
                },
                {
                    'milestone': '90% düzelme',
                    'days': half_life * 3.32,
                    'date': (datetime.now() + timedelta(days=half_life * 3.32 - last_use_days)).strftime('%Y-%m-%d') if last_use_days < half_life * 3.32 else 'Geçildi'
                },
                {
                    'milestone': 'Tam düzelme',
                    'days': full_recovery,
                    'date': (datetime.now() + timedelta(days=full_recovery - last_use_days)).strftime('%Y-%m-%d') if last_use_days < full_recovery else 'Geçildi'
                }
            ]
        }
    
    def calculate_duration_ci(self, duration_days):
        """
        Kullanım süresi güven aralığı
        """
        # %95 CI (varsayılan: ±20%)
        lower = duration_days * 0.8
        upper = duration_days * 1.2
        
        return {
            'lower_days': float(lower),
            'upper_days': float(upper),
            'lower_months': float(lower / 30),
            'upper_months': float(upper / 30)
        }
    
    def calculate_timing_ci(self, last_use_days):
        """
        Son kullanım zamanı güven aralığı
        """
        # %95 CI (varsayılan: ±30%)
        lower = max(0, last_use_days * 0.7)
        upper = last_use_days * 1.3
        
        return {
            'lower_days': float(lower),
            'upper_days': float(upper),
            'lower_date': (datetime.now() - timedelta(days=upper)).strftime('%Y-%m-%d'),
            'upper_date': (datetime.now() - timedelta(days=lower)).strftime('%Y-%m-%d')
        }
    
    def train_models(self):
        """
        Modelleri eğit (ilk kurulum için)
        """
        print("Model eğitimi başlatılıyor...")
        
        # Eğitim verisi yükle (simüle edilmiş)
        # Gerçek uygulamada: literatürden ve klinik çalışmalardan veri
        
        # Placeholder: Random Forest classifier
        from sklearn.ensemble import RandomForestClassifier, GradientBoostingRegressor
        from sklearn.preprocessing import StandardScaler
        from sklearn.decomposition import PCA
        
        self.substance_classifier = RandomForestClassifier(
            n_estimators=500,
            max_depth=20,
            min_samples_split=10,
            random_state=42
        )
        
        self.duration_estimator = GradientBoostingRegressor(
            n_estimators=300,
            max_depth=10,
            learning_rate=0.1,
            random_state=42
        )
        
        self.timing_estimator = GradientBoostingRegressor(
            n_estimators=300,
            max_depth=10,
            learning_rate=0.1,
            random_state=42
        )
        
        self.scaler = StandardScaler()
        self.pca = PCA(n_components=100)
        
        # Modelleri kaydet
        models = {
            'classifier': self.substance_classifier,
            'duration': self.duration_estimator,
            'timing': self.timing_estimator,
            'scaler': self.scaler,
            'pca': self.pca
        }
        
        with open(self.model_path, 'wb') as f:
            pickle.dump(models, f)
        
        print("✓ Modeller eğitildi ve kaydedildi")

def analyze_blood_sample(
    methylation_file,
    patient_info,
    output_dir='output'
):
    """
    Kan örneğinden madde tespiti
    
    Args:
        methylation_file: str, IDAT dosyası veya beta values CSV
        patient_info: dict, hasta bilgileri
        output_dir: str, çıktı dizini
        
    Returns:
        results: dict, analiz sonuçları
    """
    import os
    os.makedirs(output_dir, exist_ok=True)
    
    print("=" * 80)
    print("EPİCLOCK MADDE TESPİT SİSTEMİ")
    print("=" * 80)
    print(f"Hasta ID: {patient_info.get('id', 'N/A')}")
    print(f"Analiz Tarihi: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print("=" * 80)
    
    # 1. Metilasyon verisini yükle
    print("\n[1/5] Metilasyon verisi yükleniyor...")
    if methylation_file.endswith('.csv'):
        beta_values = pd.read_csv(methylation_file, index_col=0)
    else:
        # IDAT dosyası işleme (minfi/sesame kullan)
        print("  IDAT dosyası işleniyor...")
        # R script çağır veya Python implementasyonu
        beta_values = load_idat_files(methylation_file)
    
    print(f"  ✓ {beta_values.shape[0]} CpG site yüklendi")
    
    # 2. Madde tespiti
    print("\n[2/5] Madde tespiti yapılıyor...")
    detector = SubstanceDetector()
    results = detector.detect_substance(beta_values)
    print(f"  ✓ Birincil madde: {results['primary_substance']['class']}")
    print(f"  ✓ Olasılık: %{results['primary_substance']['probability']*100:.1f}")
    
    # 3. JSON kaydet
    print("\n[3/5] Sonuçlar JSON formatında kaydediliyor...")
    json_file = os.path.join(output_dir, f"{patient_info['id']}_results.json")
    with open(json_file, 'w', encoding='utf-8') as f:
        json.dump(results, f, indent=2, ensure_ascii=False)
    print(f"  ✓ JSON kaydedildi: {json_file}")
    
    # 4. PDF rapor oluştur
    print("\n[4/5] PDF raporu oluşturuluyor...")
    report_gen = SubstanceReportGenerator()
    pdf_file = os.path.join(output_dir, f"{patient_info['id']}_report.pdf")
    report_gen.generate_report(results, patient_info, pdf_file)
    print(f"  ✓ PDF kaydedildi: {pdf_file}")
    
    # 5. Özet yazdır
    print("\n[5/5] Analiz tamamlandı!")
    print("\n" + "=" * 80)
    print("ÖZET SONUÇLAR")
    print("=" * 80)
    print(f"Tespit Edilen Madde: {results['primary_substance']['class'].upper()}")
    print(f"Kullanım Süresi: {results['usage_duration']['estimated_months']:.1f} ay")
    print(f"Son Kullanım: {results['last_use']['estimated_days_ago']:.0f} gün önce")
    print(f"Kullanım Paterni: {results['usage_pattern']['description']}")
    print(f"Risk Seviyesi: {results['health_risk_assessment']['risk_category']}")
    print("=" * 80)
    
    return results

from epiclock import SubstanceDetector
import pandas as pd

# Detector başlat
detector = SubstanceDetector()

# Metilasyon verisini yükle
beta_values = pd.read_csv('data/patient_methylation.csv', index_col=0)

# Analiz çalıştır
results = detector.detect_substance(beta_values)

# Sonuçları yazdır
print(f"Tespit edilen madde: {results['primary_substance']['class']}")
print(f"Olasılık: {results['primary_substance']['probability']:.2%}")
print(f"Kullanım süresi: {results['usage_duration']['estimated_months']:.1f} ay")
print(f"Son kullanım: {results['last_use']['estimated_days_ago']:.0f} gün önce")

# Detaylı rapor
from epiclock import SubstanceReportGenerator

report_gen = SubstanceReportGenerator()
report_gen.generate_report(
    results,
    patient_info={'id': 'P-001', 'name': 'John Doe', 'age': 35},
    output_file='report.pdf'
)
